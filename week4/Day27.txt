## Day27 - Serverless Scalling with Lambda and DynamoDB
## Ansible - Deploy Serverless Lambda Functions
==========================================================
---
- name: Deploy Serverless Lambda Functions
  hosts: localhost
  become: true
  tasks:
	- name: Create Lambda function
	  command: aws lambda create-function --function-name pathnex-lambda --runtime nodejs14.x --role arn:aws:iam::3434383245:role/lambda-role --handler index.handler --zip-file fileb://lambda.zip
	  
=====================================================================
## Terraform - Create DynamoDB Table for Serverless Application
=====================================================================
resource "aws_dynamodb_table" "pathnex-table" {
	name = "pathnex-table"
	has_key = "id"
	billing_mode = "PAY_PER_REQUEST"
	attribute {
		name = "id"
		type = "s"
	}
}


=====================================================================
## Kubernetes - Deploy Serverless Using Kubeless
=====================================================================
apiVersion: kubeless.io/v1beta1
kind: Function
metadata:
	name: pathnex-lambda
	namespace: default
spec:
	runtime: nodejs14
	hadler: index.handler
	function: |
		module.exports = function(event, context) {
			context.succeed('Hello World');
		};
		
====================================================================
## Jenkinsfile - Lambda Function Deployment Pipeline
====================================================================

pipeline {
	agent any
	stage {
		stage('Build') {
			steps {
				echo 'Building Lambda function package..'
				sh 'zip -r lambda.zip lambda/'
			}
		}
		stage('Deploy to Lambda') {
			steps {
				script {
					sh 'aws lambda update-function-code --function-name pathnex-lambda --zip-file fileb://lambda.zip'
				}
			}
		}
	}
}
